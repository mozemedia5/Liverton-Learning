rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isAuthenticated() {
      return request.auth != null;
    }
    function currentUserId() {
      return request.auth.uid;
    }
    function currentUserData() {
      return get(/databases/$(database)/documents/users/$(currentUserId())).data;
    }
    function isPlatformAdmin() {
      return isAuthenticated() && currentUserData().role == 'platform_admin';
    }
    function isSchoolAdmin() {
      return isAuthenticated() && currentUserData().role == 'school_admin';
    }
    function isTeacher() {
      return isAuthenticated() && currentUserData().role == 'teacher';
    }
    function isParent() {
      return isAuthenticated() && currentUserData().role == 'parent';
    }
    function isStudent() {
      return isAuthenticated() && currentUserData().role == 'student';
    }
    function currentUserSchoolId() {
      return currentUserData().schoolId;
    }
    match /users/{userId} {
      allow read: if isPlatformAdmin();
      allow write: if isPlatformAdmin();
      allow read, write: if isAuthenticated() && currentUserId() == userId;
      allow read: if isSchoolAdmin() &&
        (resource.data.schoolId == currentUserSchoolId() ||
         resource.data.role == 'teacher' && resource.data.schoolId == currentUserSchoolId() ||
         resource.data.role == 'student' && resource.data.schoolId == currentUserSchoolId());
      allow read: if isTeacher() &&
        resource.data.role == 'student' &&
        resource.data.schoolId == currentUserSchoolId();
      allow read: if isTeacher() &&
        resource.data.role == 'teacher' &&
        resource.data.schoolId == currentUserSchoolId();
      allow read: if isParent() &&
        resource.data.role == 'student' &&
        currentUserData().linkedStudentIds != null &&
        userId in currentUserData().linkedStudentIds;
      allow create: if isAuthenticated() && currentUserId() == userId;
    }
    match /schools/{schoolId} {
      allow read, write: if isPlatformAdmin();
      allow read: if isAuthenticated();
      allow write: if isSchoolAdmin() &&
        currentUserData().schoolId == schoolId;
    }
    match /courses/{courseId} {
      allow read, write: if isPlatformAdmin();
      allow read: if isSchoolAdmin() &&
        resource.data.schoolId == currentUserSchoolId();
      allow create: if isSchoolAdmin() &&
        request.resource.data.schoolId == currentUserSchoolId();
      allow read: if isTeacher() &&
        (resource.data.teacherId == currentUserId() ||
         resource.data.schoolId == currentUserSchoolId());
      allow create: if isTeacher();
      allow update: if isTeacher() &&
        resource.data.teacherId == currentUserId();
      allow read: if isStudent() &&
        currentUserId() in resource.data.enrolledStudents;
      allow read: if isParent();
    }
    match /announcements/{announcementId} {
      allow read, write: if isPlatformAdmin();
      allow create: if isSchoolAdmin() &&
        (request.resource.data.targetSchoolId == currentUserSchoolId() ||
         request.resource.data.targetSchoolId == null);
      allow read: if isSchoolAdmin() &&
        (resource.data.targetSchoolId == currentUserSchoolId() ||
         resource.data.targetSchoolId == null);
      allow read: if isAuthenticated();
      allow write: if isPlatformAdmin() || isSchoolAdmin();
    }
    match /chats/{chatId} {
      function isChatParticipant(chatData) {
        return isAuthenticated() && currentUserId() in chatData.participants;
      }

      allow read: if isPlatformAdmin();
      allow read: if isChatParticipant(resource.data);
      allow create: if isAuthenticated() &&
        currentUserId() in request.resource.data.participants &&
        request.resource.data.participants.size() >= 2 &&
        request.resource.data.participants.size() <= 50;
      allow update: if isChatParticipant(resource.data) &&
        request.resource.data.participants == resource.data.participants;
      allow delete: if isChatParticipant(resource.data);

      match /messages/{messageId} {
        function parentChat() {
          return get(/databases/$(database)/documents/chats/$(chatId));
        }
        function isMessageParticipant() {
          return isAuthenticated() && currentUserId() in parentChat().data.participants;
        }

        allow read: if isMessageParticipant();
        allow create: if isMessageParticipant() &&
          request.resource.data.chatId == chatId &&
          request.resource.data.senderId == currentUserId();
        allow update: if isMessageParticipant() &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['readBy']) &&
          currentUserId() in request.resource.data.readBy;
        allow delete: if isMessageParticipant();
      }
    }
    match /documents/{documentId} {
      allow read, write: if isPlatformAdmin();
      allow get: if resource.data.visibility == 'public';
      allow get, list: if isAuthenticated() &&
        resource.data.visibility in ['public', 'internal'];
      allow get, list: if isAuthenticated() &&
        (resource.data.ownerId == currentUserId() ||
         currentUserId() in resource.data.sharedWith);
      allow create: if isAuthenticated() &&
        request.resource.data.ownerId == currentUserId();
      allow update: if isAuthenticated() &&
        (resource.data.ownerId == currentUserId() ||
         currentUserId() in resource.data.sharedWith);
      allow delete: if isAuthenticated() &&
        resource.data.ownerId == currentUserId();
      match /{sub=**} {
        allow read, write: if isPlatformAdmin() ||
          (isAuthenticated() &&
            (get(/databases/$(database)/documents/documents/$(documentId)).data.ownerId == currentUserId() ||
             currentUserId() in get(/databases/$(database)/documents/documents/$(documentId)).data.sharedWith));
      }
    }
    match /quizzes/{quizId} {
      allow read, write: if isPlatformAdmin();
      allow read, write: if isTeacher();
      allow read: if isStudent();
      allow read: if isParent();
      allow read: if isSchoolAdmin();
    }
    match /attendance/{recordId} {
      allow read, write: if isPlatformAdmin();
      allow read, write: if isSchoolAdmin() &&
        resource.data.schoolId == currentUserSchoolId();
      allow read, write: if isTeacher() &&
        resource.data.schoolId == currentUserSchoolId();
      allow read: if isStudent() &&
        resource.data.studentId == currentUserId();
      allow read: if isParent();
    }
    match /payments/{paymentId} {
      allow read, write: if isPlatformAdmin();
      allow read: if isAuthenticated() &&
        resource.data.userId == currentUserId();
      allow create: if isAuthenticated() &&
        request.resource.data.userId == currentUserId();
      allow read: if isSchoolAdmin();
    }
    match /dashboardBanners/{bannerId} {
      allow read: if isAuthenticated();
      allow write: if isPlatformAdmin();
    }
    match /shorts/{shortId} {
      allow read: if isAuthenticated();
      allow create: if isTeacher() &&
        request.resource.data.teacherId == currentUserId() &&
        ((request.resource.data.get('courseId', null) != null && request.resource.data.get('lessonId', null) == null) ||
         (request.resource.data.get('courseId', null) == null && request.resource.data.get('lessonId', null) != null));
      allow update: if isTeacher() && resource.data.teacherId == currentUserId();
      allow update: if isAuthenticated() &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['likes', 'views']);
      allow delete: if isTeacher() && resource.data.teacherId == currentUserId();
    }
    match /events/{eventId} {
      allow read, write: if isPlatformAdmin();
      allow get, list: if isAuthenticated() &&
        resource.data.visibility == 'public';
      allow get, list: if isAuthenticated() &&
        resource.data.visibility == 'school' &&
        resource.data.schoolId == currentUserSchoolId();
      allow get, list: if isAuthenticated() &&
        resource.data.createdBy == currentUserId();
      allow create: if isAuthenticated() &&
        request.resource.data.createdBy == currentUserId();
      allow update, delete: if isAuthenticated() &&
        resource.data.createdBy == currentUserId();
    }
